  function ellipse(centerX, centerY, radiusX, radiusY, value = 1) {
    let xStart = Math.max(0, Math.floor(centerX - radiusX));
    let xEnd = Math.min(tw, Math.ceil(centerX + radiusX));
    let yStart = Math.max(0, Math.floor(centerY - radiusY));
    let yEnd = Math.min(th, Math.ceil(centerY + radiusY));
  
    for (let y = yStart; y < yEnd; y++) {
      for (let x = xStart; x < xEnd; x++) {
        
        let dx = x - centerX;
        let dy = y - centerY;

        let dv = (dx * dx) / (radiusX * radiusX) + (dy * dy) / (radiusY * radiusY);
        if (dv <= 1) {
          let index = y * tw + x;
          tiles[index] = value;
        }
      }
    }
  }
  function bezier(x0, y0, x1, y1, x2, y2, steps = 100, value = 1) {
    for (let i = 0; i <= steps; i++) {
      // t is the percentage of the way along the curve (0.0 to 1.0)
      let t = i / steps;
      
      // Quadratic Bezier formula
      let invT = 1 - t;
      let x = invT * invT * x0 + 2 * invT * t * x1 + t * t * x2;
      let y = invT * invT * y0 + 2 * invT * t * y1 + t * t * y2;
  
      // Convert floating point coordinates to integer grid coordinates
      let gridX = Math.floor(x);
      let gridY = Math.floor(y);
  
      // Safety check: ensure coordinates are within the grid bounds
      if (gridX >= 0 && gridX < tw && gridY >= 0 && gridY < th) {
        let index = gridY * tw + gridX;
        tiles[index] = value;
      }
    }
  }


  function leaveOutline() {
  let nextTiles = new Array(tiles.length).fill(0);

  for (let y = 1; y < th - 1; y++) {
    for (let x = 1; x < tw - 1; x++) {
      let i = y * tw + x;
      if (tiles[i] === 0) continue;

      let hasEmptyNeighbor = 
        tiles[i - 1] === 0 ||
        tiles[i + 1] === 0 ||
        tiles[i - tw] === 0 ||
        tiles[i + tw] === 0;
      if (hasEmptyNeighbor) {
        nextTiles[i] = tiles[i];
      }
    }
  }
  tiles = nextTiles;
  };

function rect(x, y, w, h, value = 1) {
  if (w < 0) {x += w; w = -w;}
  x = Math.round(x);
  y = Math.round(y);
  let xStart = Math.max(0, Math.floor(x));
  let xEnd = Math.min(tw, Math.ceil(x + w));
  let yStart = Math.max(0, Math.floor(y));
  let yEnd = Math.min(th, Math.ceil(y + h));

  for (let ty = yStart; ty < yEnd; ty++) {
    for (let tx = xStart; tx < xEnd; tx++) {
      let index = ty * tw + tx;
      if (overwriteOnly && tiles[index] == 0) {continue;}
      tiles[index] = value;
    }
  }
}

function line(x0, y0, x1, y1, value = 1) {
  x0 = Math.floor(x0);
  y0 = Math.floor(y0);
  x1 = Math.floor(x1);
  y1 = Math.floor(y1);

  let dx = Math.abs(x1 - x0);
  let dy = Math.abs(y1 - y0);
  let sx = (x0 < x1) ? 1 : -1;
  let sy = (y0 < y1) ? 1 : -1;
  let err = dx - dy;

  while (true) {
    // Safety check: stay within grid bounds
    if (x0 >= 0 && x0 < tw && y0 >= 0 && y0 < th) {
      tiles[y0 * tw + x0] = value;
    }

    if (x0 === x1 && y0 === y1) break;
    
    let e2 = 2 * err;
    if (e2 > -dy) {
      err -= dy;
      x0 += sx;
    }
    if (e2 < dx) {
      err += dx;
      y0 += sy;
    }
  }
}

function thickLine(x0, y0, x1, y1, thickness, value = 1) {
  x0 = Math.round(x0);
  y0 = Math.round(y0);
  x1 = Math.round(x1);
  y1 = Math.round(y1);
  // Bresenham's algorithm remains the same for the pathing
  let dx = Math.abs(x1 - x0);
  let dy = Math.abs(y1 - y0);
  let sx = (x0 < x1) ? 1 : -1;
  let sy = (y0 < y1) ? 1 : -1;
  let err = dx - dy;

  let radius = thickness / 2;

  while (true) {
    // Instead of one pixel, draw a filled circle/square "brush"
    for (let iy = -Math.floor(radius); iy <= Math.ceil(radius); iy++) {
      for (let ix = -Math.floor(radius); ix <= Math.ceil(radius); ix++) {
        // Distance check for a circular brush (prevents blocky corners)
        if (ix * ix + iy * iy <= radius * radius) {
          let px = Math.floor(x0 + ix);
          let py = Math.floor(y0 + iy);
          
          if (px >= 0 && px < tw && py >= 0 && py < th) {
            tiles[py * tw + px] = value;
          }
        }
      }
    }

    if (x0 === x1 && y0 === y1) break;
    let e2 = 2 * err;
    if (e2 > -dy) { err -= dy; x0 += sx; }
    if (e2 < dx) { err += dx; y0 += sy; }
  }
}

function lineAngle(x, y, len, angleXY, angleZ = 0, value = 1) {
  let distXY = len * Math.cos(angleZ);
  
  let x1 = x + distXY * Math.cos(angleXY);
  let y1 = y + distXY * Math.sin(angleXY);

  line(x, y, x1, y1, value);
}

}

function patternLine(x0, y0, x1, y1, thickness, palette, stripeWidth = 2) {
  
  x0 = Math.round(x0);
  y0 = Math.round(y0);
  x1 = Math.round(x1);
  y1 = Math.round(y1);
  let dx = Math.abs(x1 - x0);
  let dy = Math.abs(y1 - y0);
  let sx = (x0 < x1) ? 1 : -1;
  let sy = (y0 < y1) ? 1 : -1;
  let err = dx - dy;

  let radius = thickness / 2;

  //let iter = 0;
  let patternOrgs = [];
  while (true) {
    for (let iy = -Math.floor(radius); iy <= Math.ceil(radius); iy++) {
      for (let ix = -Math.floor(radius); ix <= Math.ceil(radius); ix++) {
        if (ix * ix + iy * iy <= radius * radius) {
          let px = Math.floor(x0 + ix);
          let py = Math.floor(y0 + iy);

          let canDo = false;
          if ((ix - 1) * (ix - 1) + iy * iy > radius * radius) {
            canDo = true;
          }
          if (px >= 0 && px < tw && py >= 0 && py < th) {
            // THE MATH: (px + py) creates the diagonal. 
            // Dividing by stripeWidth controls thickness.
            // Modulo palette.length cycles through colors.
            let colorIndex = Math.abs(px - ((py)/2)) % palette.length;
            if (colorIndex == 2 && canDo) {
              colorIndex = 2;
            } else {
              colorIndex = 0;
            }
            tiles[py * tw + px] = palette[colorIndex];
          }
        }
      }
    }

    if (x0 === x1 && y0 === y1) break;
    let e2 = 2 * err;
    if (e2 > -dy) { err -= dy; x0 += sx; }
    if (e2 < dx) { err += dx; y0 += sy; }
  }
}
