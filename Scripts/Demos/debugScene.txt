function getShaderVar(str, programID = 0) {
  let leStr = programID + str;
  if (attributeLocations[leStr] == null) {
    if (str[0] == "a") {
      attributeLocations[leStr] = gl.getAttribLocation(program, str);
    } else if (str[0] == "u") {
      attributeLocations[leStr] = gl.getUniformLocation(program, str);
    }
  }
  return attributeLocations[leStr];
}

function getMouseRayDir(mouseX, mouseY, canvasWidth, canvasHeight, fovDeg, aspect, yawDeg, pitchDeg) {
    const ndcX = (2 * mouseX / canvasWidth - 1);
    const ndcY = (1 - 2 * mouseY / canvasHeight);

    const fovY = degToRad(fovDeg);
    const tanFovY = Math.tan(fovY / 2);
    const tanFovX = tanFovY * aspect;

    // Step 1: camera-space ray
    let dir = [
        ndcX * tanFovX,
        -ndcY * tanFovY,
        -1.0
    ];

    dir = vec3.normalize(dir);

    // Step 2: convert yaw/pitch to radians
    const yaw = degToRad(yawDeg);
    const pitch = degToRad(pitchDeg);

    // Step 3: apply rotations (camera to world)
    // Rotation around Y (yaw), then X (pitch)
    const cosYaw = Math.cos(yaw);
    const sinYaw = Math.sin(yaw);
    const cosPitch = Math.cos(pitch);
    const sinPitch = Math.sin(pitch);

    // Rotate by pitch (X-axis)
    let dx = dir[0];
    let dy = dir[1] * cosPitch - dir[2] * sinPitch;
    let dz = dir[1] * sinPitch + dir[2] * cosPitch;

    // Rotate by yaw (Y-axis)
    const worldX = dx * cosYaw + dz * sinYaw;
    const worldY = dy;
    const worldZ = -dx * sinYaw + dz * cosYaw;

    return vec3.normalize([worldX, worldY, worldZ]);
}
