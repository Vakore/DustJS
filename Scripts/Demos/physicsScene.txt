    } else if (str[0] == "u") {
      attributeLocations[leStr] = gl.getUniformLocation(program, str);
    }
  }
  return attributeLocations[leStr];
}

function raycastModels(objs, rayOrigin, rayDirection, maxDistance = Infinity) {
  let closestHit = null;
  for (let i = 0; i < objs.length; i++) {
    let thisHit = raycastModel(i, objs[i].model, objs[i].matrix, objs[i].scaleFactor, rayOrigin, rayDirection, maxDistance);
    if (!closestHit || thisHit != null && thisHit.distance <= closestHit.distance) {
      closestHit = thisHit;
    }
  }
  return closestHit;
}

function raycastModel(index, model, worldMatrix, scaler, rayOriginWorld, rayDirectionWorld, maxDistance = Infinity) {
    rayDirectionWorld = vec3.normalize(rayDirectionWorld);
    
    const verts = model.verts;
    const tris = model.tris;
    
    let closestHit = null;
    let closestDistSq = maxDistance * maxDistance;
    
    const invMatrix = m4.inverse(worldMatrix);
    let rayOriginLocal = m4.transformPoint(invMatrix, rayOriginWorld);
    let rayDirectionLocal = m4.transformDirection(invMatrix, rayDirectionWorld);
    
    const dirLen = vec3.length(rayDirectionLocal);
    const rayDirectionLocalNormalized = vec3.normalize(rayDirectionLocal);
    const maxDistanceLocal = maxDistance * dirLen;

    for (let i = 0; i < tris.length; i += 3) {
        const idxA = tris[i] * 3;
        const idxB = tris[i + 1] * 3;
        const idxC = tris[i + 2] * 3;
        
        const v0 = [verts[idxA], verts[idxA+1], verts[idxA+2]];
        const v1 = [verts[idxB], verts[idxB+1], verts[idxB+2]];
        const v2 = [verts[idxC], verts[idxC+1], verts[idxC+2]];
        
        // Moller–Trumbore ray-triangle intersection
        const edge1 = vec3.subtract(v1, v0);
        const edge2 = vec3.subtract(v2, v0);
        const h = vec3.cross(rayDirectionLocalNormalized, edge2);
        const a = vec3.dot(edge1, h);
        
        // Parallel check (avoid division by zero)
        if (Math.abs(a) < 0.000001) continue;
        
        const f = 1.0 / a;
        const s = vec3.subtract(rayOriginLocal, v0);
        const u = f * vec3.dot(s, h);
        if (u < 0.0 || u > 1.0) continue;
        
        const q = vec3.cross(s, edge1);
        const v = f * vec3.dot(rayDirectionLocalNormalized, q);
        if (v < 0.0 || u + v > 1.0) continue;
        
        const tLocal = f * vec3.dot(edge2, q); // distance in local space along normalized ray
        if (tLocal <= 0.000001 || tLocal >= maxDistanceLocal) continue;
        
        // Convert local distance to world distance
        const tWorld = tLocal / dirLen;
        
        if (tWorld < Math.sqrt(closestDistSq)) {
            closestDistSq = tWorld * tWorld;
            
            const localHitPoint = vec3.add(rayOriginLocal, vec3.scale(rayDirectionLocalNormalized, tLocal));
            
            // Compute triangle normal (assuming counter-clockwise winding)
            let localNormal = vec3.cross(edge1, edge2);
            localNormal = vec3.normalize(localNormal);
            
            const worldHitPoint = m4.transformPoint(worldMatrix, localHitPoint);
            let worldNormal = m4.transformDirection(worldMatrix, localNormal);
            worldNormal = vec3.normalize(worldNormal);
            
            // Make sure normal points towards ray origin, flip if necessary
            if (vec3.dot(worldNormal, rayDirectionWorld) > 0) {
                worldNormal = vec3.scale(worldNormal, -1);
            }
            
            closestHit = {
                point: worldHitPoint,
                distance: tWorld, // Use world distance
                normal: worldNormal,
                triangleIndex: i / 3,
                objId: index,
                barycentric: [1 - u - v, u, v],
                matrix:worldMatrix
            };
        }
    }
    
    return closestHit;
}

    //ignore edge collision for walls. This doesn't work with seams tho
    /*if (isGround >= -0.7 && Math.abs(collisionNormal[1]) >= 0.1) {
      
      return;
    }*/

    //For some displacementWorld can become NaN on surfaces with many polygons,
    //like the tippy-top of spheres. Also oddly simply ignoring these collisions appears
    //to have no side effects? This scares meand is bound to break methinks...
    //Yup, you can clip through it if you jump correctly. This needs fixing...
    if (isNaN(displacementWorld[0])) {console.log("bruh");return;}
    if (isGround <= 0.7) {
      entity.pos.x += displacementWorld[0];
      entity.pos.y += displacementWorld[1];
      entity.pos.z += displacementWorld[2];
    } else {
      let pushMag = vec3.length(displacementWorld);
      //alert(pushMag);
      entity.pos.x += pushMag * entity.gravityNormal[0];
      entity.pos.y += pushMag * entity.gravityNormal[1];
      entity.pos.z += pushMag * entity.gravityNormal[2];
    }
 
    //For those who come after
    worldPos = [entity.pos.x, entity.pos.y, entity.pos.z];
    localPos = m4.transformPoint(invMatrix, worldPos);
 
    if (isGround > 0.7) {
      //console.log(isGround);
      entity.groundNormal = worldGroundNormal;
    } else {
      entity.wallNormal = worldGroundNormal;
    }
 
    // Velocity reflection in world space (inelastic, absorbs y if falling; add restitution if bouncy)
    let worldVel = [entity.vel.x, entity.vel.y, entity.vel.z];

    if (entity.rollable) {
      let dot = vec3.dot(worldVel, worldGroundNormal);
      if (dot < 0) {
        entity.vel.x -= dot * worldGroundNormal[0];
        entity.vel.y -= dot * worldGroundNormal[1];
        entity.vel.z -= dot * worldGroundNormal[2];
      }
    } else {
      if (isGround > 0.7){
        if (entity.relVel[1] < 0) entity.relVel[1] = 0;
      } else if (isGround > 0) {
      }
      
      //console.log("coll: " + index + "_" + j);
      entity.carryMatrices[index] = worldMatrix;
    }
  }
}

function sphereEntity(x, y, z, r) {
  this.rotMatrix = m4.identity();
  this.pos = new Vec3(x, y, z);
  this.ppos = new Vec3(x, y, z);
  this.vel = new Vec3(0, 0, 0);
  this.radius = r;
  this.grounded = false;
  this.gravityNormal = [0, -1, 0];
  this.groundNormal = [0, 1, 0];
  this.rollable = true;
}


sphereEntity.prototype.run = function() {
  if (!this.grounded) {
    this.gravityNormal = [0, -1, 0];
    this.groundNormal = [0, 1, 0];
  }
  this.vel.z = 0.1;
  let MAX_SPEED = 0.4;
  let spd = Math.hypot(this.vel.x, this.vel.z);
  if (spd > MAX_SPEED) {
      this.vel.x = (this.vel.x / spd) * MAX_SPEED;
      this.vel.z = (this.vel.z / spd) * MAX_SPEED;
  }
  if (!this.grounded) {
    this.vel.y -= 0.01;
  }
  this.grounded = false;
  this.ppos.x = this.pos.x;
  this.ppos.y = this.pos.y;
  this.ppos.z = this.pos.z;
  this.pos.x += this.vel.x;
  this.pos.y += this.vel.y;
  this.pos.z += this.vel.z;

  for (let i = 0; i < 3; i++) {
    meshCollisions(this, scene.objs);
  }
  
  let rcast = raycastModels(scene.objs, [this.pos.x, this.pos.y, this.pos.z], vec3.invert(this.groundNormal), this.radius + spd + 0.1);
  if (rcast && this.vel.y <= 0 && this.vel.y >= -0.01) {
    //rcast.normal = vec3.normalize(rcast.normal)
    this.grounded = true;
    this.pos.x -= (rcast.distance - this.radius) * rcast.normal[0] * this.gravityNormal[0];
    this.pos.y -= (rcast.distance - this.radius) * rcast.normal[1] * this.gravityNormal[1];
    this.pos.z -= (rcast.distance - this.radius) * rcast.normal[2] * this.gravityNormal[2];
    //console.log(rcast.normal);
    this.vel.y = 0;
    //console.log("e");
  }
  
  let dx = -this.vel.x;
  let dz = -this.vel.z;
  let dist = Math.hypot(dx, dz);
  if (dist > 0) {
      let theta = dist / this.radius;
      let dir_x = dx / dist;
      let dir_z = dz / dist;
      let axis = [-dir_z, 0, dir_x];
      let small_rot = m4.axisRotation(axis, theta);
      this.rotMatrix = m4.multiply(small_rot, this.rotMatrix);
  }
};

sphereEntity.prototype.draw = function() {

  m = m4.translate(m, this.pos.x, this.pos.y, this.pos.z);
  m = m4.multiply(m, this.rotMatrix);
  m = m4.scale(m, this.radius, this.radius, this.radius);
  sphereModel.prep(planeTexture, false);
  sphereModel.draw(m, drawMode ? gl.TRIANGLES : gl.LINE_STRIP);
}
