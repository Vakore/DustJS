let circleModel = new model();
let segments = 32; // The resolution of the circle
let radius = 100;  // Matching your plane's scale
let texScale = 10; // Matching your plane's texture scale

// 1. Define Vertices, Texcoords, and Colors
// Start with the center point
circleModel.verts.push(0.0, 0.0, 0.0);
circleModel.texcoords.push(0.5 * texScale, 0.5 * texScale);
circleModel.colors.push(255, 255, 255);

for (let i = 0; i <= segments; i++) {
    let theta = (i / segments) * Math.PI * 2;
    let x = Math.cos(theta);
    let z = Math.sin(theta);

    // Vertices (Position)
    circleModel.verts.push(x * radius, 0.0, z * radius);

    // Texture Coordinates (Mapping -1 to 1 range to 0 to 1 range)
    let u = (x * 0.5 + 0.5) * texScale;
    let v = (z * 0.5 + 0.5) * texScale;
    circleModel.texcoords.push(u, v);

    // Colors
    circleModel.colors.push(255, 255, 255);
}

// 2. Define Triangles (Indices)
// Connect the center (index 0) to two adjacent points on the perimeter
for (let i = 1; i <= segments; i++) {
    circleModel.tris.push(0, i, i + 1);
}

// 3. Setup Bone Data (Standard rigging for a static object)
for (let i = 0; i < circleModel.verts.length / 3; i++) {
    circleModel.boneIDs.push(0, 0, 0, 0);
    circleModel.boneWeights.push(1.0, 0.0, 0.0, 0.0);
}

// 4. Initialize Bone Hierarchy
circleModel.bones = [{
    children: [], 
    parent: -1, 
    locoldPos: m4.identity(), 
    localPos: m4.translation(0.0, 0.0, 0.0), 
    curPos: m4.identity(), 
    binvPos: m4.identity()
}];

// 5. Finalize
computeBindMatrices(circleModel, 0, m4.identity());
circleModel.typify();
