let yellowColor = {
  colors:[]
};
let cubeModel = {
  verts:new Float32Array([-0.5,-0.5,-0.5, 0.5,-0.5,-0.5,
         -0.5,-0.5,0.5, 0.5,-0.5,0.5,
         -0.5,0.5,-0.5, 0.5,0.5,-0.5,
         -0.5,0.5,0.5, 0.5,0.5,0.5]),//x, y, z
  colors:[],//r, g, b
  texCoords:[],//x, y
  tris:new Uint16Array([0, 1, 2, 2, 1, 3,
                        4, 6, 5, 6, 7, 5,
                        2, 3, 6, 6, 3, 7,
                        0, 4, 1, 4, 5, 1,
                        0, 2, 4, 4, 2, 6,
                        1, 5, 3, 3, 5, 7]),//indices[0, 1, 2]
  bones:[],
  boneIDs:[],
  boneWeights:[]
};

for (let i = 0; i < cubeModel.verts.length / 3; i++) {
  cubeModel.colors.push(255,255,255);
  yellowColor.colors.push(255,255,0);
  cubeModel.boneIDs.push(0,0,0,0);
  cubeModel.boneWeights.push(1.0,0.0,0.0,0.0);
}
yellowColor.colors = new Uint8Array(yellowColor.colors);
cubeModel.colors = new Uint8Array(cubeModel.colors);
cubeModel.boneIDs = new Uint8Array(cubeModel.boneIDs);
cubeModel.boneWeights = new Float32Array(cubeModel.boneWeights);

cubeModel.bones = [
{children:[], curPos:m4.identity(), binvPos:m4.identity()},
];

cubeModel.bones[0].curPos = m4.translate(cubeModel.bones[0].curPos, 0.0, 0.0, 0.0);
for (let i = 0; i < cubeModel.bones.length; i++) {
  cubeModel.bones[i].binvPos = m4.inverse(cubeModel.bones[i].curPos);
}
