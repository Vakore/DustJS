  let torsoRingR = [];

  let midpoint = 6;

        vordasModel.boneWeights.push(1.0,0.0,0.0,0.0);

  //earclipRings(vordasModel, vordasModel.rings[vordasModel.rings.length - 1], []);
  
      for (let i = pvordVert; i < vordasModel.verts.length / 3; i++) {

        vordasModel.boneWeights.push(1.0,0.0,0.0,0.0);
      }

  }
  
  //vordasModel.rings.push([], [], [], [], [], []);

  /*yp = [2.0, 2.5, 3.5, 5.75, 8.0, 8.5, 9.0, 10.0, 10.5, 11];
