/**
 * Loads a script tag into the DOM and returns a Promise.
 */
function loadScript(source) {
  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = source;
    // Essential: Script won't load unless it's in the document
    script.async = false; // Ensures they execute in order if added simultaneously

    script.onload = () => {
      console.log(`%c Loaded: ${source}`, "color: #4CAF50; font-weight: bold;");
      resolve();
    };

    script.onerror = () => {
      reject(new Error(`Failed to load script: ${source}`));
    };

    document.head.appendChild(script);
  });
}

/**
 * Main Engine Entry Point
 */
async function initEngine() {
  console.log("Engine: Initializing sequence...");

  try {
    // Loop through the array and wait for each script to finish
    for (const path of scriptsToLoad) {
      await loadScript(path);
    }

  } catch (error) {
    console.error("Engine: Critical Failure during initialization:");
    console.error(error);
  }
}

// Start the process
initEngine();
