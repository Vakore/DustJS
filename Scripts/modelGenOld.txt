}

function connectRings(model, ringA, ringB, offsetA = 0, offsetB = 0) {
  let m = ringA.length;
  let n = ringB.length;

  let i = 0, j = 0;
  /*while (i < m) {
    model.tris.push(ringA[i], ringA[(i + 1) % m], ringB[i]);
    model.tris.push(ringB[i], ringA[(i + 1) % m], ringB[(i + 1) % m]);
    i++;
  }*/
  let currA = offsetA % m, currB = offsetB % n;

  while (i < m || j < n) {
    let nextA = (currA + 1) % m;
    let nextB = (currB + 1) % n;

    let tA = (i + 1) / m;
    let tB = (j + 1) / n;

    if ((i < m && tA <= tB) || j >= n) {
      model.tris.push(ringA[currA], ringA[nextA], ringB[currB]);
      currA = nextA;
      i++;
    } else {
      model.tris.push(ringA[currA], ringB[nextB], ringB[currB]);
      currB = nextB;
      j++;
    }
  }
}

for (let i = 0; i < springVerts.length; i += 3) {
  let closestBones = [];
  let boneDists = [];
  for (let j = 0; j < vordasModel.bones.length; j++) {
    closestBones.push(j);
    let curDist = Math.sqrt((springVerts[i]-vordasModel.bones[j].curPos[12])*(springVerts[i]-vordasModel.bones[j].curPos[12]) +
              (springVerts[i+1]-vordasModel.bones[j].curPos[13])*(springVerts[i+1]-vordasModel.bones[j].curPos[13]) +
              (springVerts[i+2]-vordasModel.bones[j].curPos[14])*(springVerts[i+2]-vordasModel.bones[j].curPos[14]));
    boneDists.push(curDist);
  }

  let ret = [boneDists, closestBones];
  for (let j = 1; j < ret[0].length; j++) {
    let k = j;
    while (k > 0 && ret[0][k - 1] > ret[0][k]) {
      let temp = ret[0][k - 1];
      let temp2 = ret[1][k - 1];
      ret[0][k - 1] = ret[0][k];
      ret[1][k - 1] = ret[1][k];
      ret[0][k] = temp;
      ret[1][k] = temp2;
      k--;
    }
  }
  
  vordasModel.boneIDs.push(ret[1][0], ret[1][1], ret[1][2], ret[1][3]);
  let sum = 0;
  for (let j = 0; j < 4; j++) {
    ret[0][j] = (1 / ret[0][j]);
    if (ret[0][j] < 0.1) {ret[0][j] = 0.0;}
    if (ret[0][j] > 1) {ret[0][j] = 1.0;}
    sum += ret[0][j];
  }
  for (let j = 0; j < 4; j++) {
    ret[0][j] /= sum;
  }
  vordasModel.boneWeights.push(ret[0][0], ret[0][1], ret[0][2], ret[0][3]);
}
